-- Drop existing tables to ensure a clean setup
DROP TABLE IF EXISTS stock_data, stocks, watchlists CASCADE;

-- Create watchlists table
CREATE TABLE watchlists (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL
);

-- Create stocks table
CREATE TABLE stocks (
    id SERIAL PRIMARY KEY,
    watchlist_id INTEGER REFERENCES watchlists(id) ON DELETE CASCADE,
    symbol VARCHAR(10) NOT NULL,
    UNIQUE (watchlist_id, symbol)
);

-- Create stock_data table
CREATE TABLE stock_data (
    stock_id INTEGER PRIMARY KEY REFERENCES stocks(id) ON DELETE CASCADE,
    symbol VARCHAR(10) NOT NULL,
    close NUMERIC,
    open NUMERIC,
    high NUMERIC,
    low NUMERIC,
    volume BIGINT,
    trend VARCHAR(10),
    macd_signal VARCHAR(10),
    mini_news TEXT,
    news_items JSONB,
    change_percent NUMERIC,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create indexes for performance
CREATE INDEX idx_stocks_symbol ON stocks(symbol);
CREATE INDEX idx_stocks_watchlist_id ON stocks(watchlist_id);
CREATE INDEX idx_stock_data_symbol ON stock_data(symbol);
CREATE INDEX idx_stock_data_trend ON stock_data(trend);
CREATE INDEX idx_stock_data_volume ON stock_data(volume);
CREATE INDEX idx_stock_data_created_at ON stock_data(created_at);
